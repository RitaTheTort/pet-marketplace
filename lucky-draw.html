function spinWheel() { if (!freeSpinsToday) { alert('คุณใช้สิทธิ์หมุนฟรีของวันนี้ไปแล้วครับ! (สามารถกดปุ่มรีเซ็ตด้านล่างเพื่อทดสอบเพิ่มได้)'); return; } const btn = document.getElementById('spinBtn'); btn.disabled = true; // 1. สุ่มเลือก Index ของช่องตรงๆ (0 ถึง 19) const targetIndex = Math.floor(Math.random() * wheelValues.length); const targetToken = wheelValues[targetIndex]; // 2. คำนวณองศาแบบล็อกเป้าแม่นยำ (ใช้มุมบวกสะสมเพื่อให้หมุนตามเข็มไปหยุดที่ช่องเป้าหมายพอดี) const sliceAngle = 360 / 20; // 18 องศาต่อช่อง const targetCenterAngle = (targetIndex * sliceAngle) + (sliceAngle / 2); // ให้หมุนไปข้างหน้า 8 รอบเต็ม + ระยะมุมที่เหลือไปยังช่องเป้าหมาย totalSpins += 8; const finalAngle = (360 * totalSpins) + (360 - targetCenterAngle); const wheel = document.getElementById('wheel'); wheel.style.setProperty('--final-rotate', `${finalAngle}deg`); // 3. แสดงผลลัพธ์หลังจากวงล้อหยุดหมุน (4 วินาที) setTimeout(() => { userTokens += targetToken; localStorage.setItem('exotic_tokens', userTokens); localStorage.setItem('exotic_free_spin', 'used'); freeSpinsToday = false; document.getElementById('userTokenCount').innerText = userTokens; updateSpinButtonUI(); btn.disabled = false; alert(`🎉 ยินดีด้วยครับ! คุณสุ่มได้ ${targetToken} Token! บันทึกลงกระเป๋าเรียบร้อยแล้ว`); }, 4000); }
