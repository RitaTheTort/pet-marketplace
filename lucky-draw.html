// เรียงลำดับตามเข็มนาฬิกา (เริ่มจาก 0 องศาด้านบนสุด หมุนตามเข็ม) const wheelValues = [1, 5, 1, 2, 1, 3, 1, 2, 1, 2, 1, 5, 1, 2, 1, 3, 1, 2, 1, 3]; function spinWheel() { if (!freeSpinsToday) { alert('คุณใช้สิทธิ์หมุนฟรีของวันนี้ไปแล้วครับ! (สามารถกดปุ่มรีเซ็ตด้านล่างเพื่อทดสอบเพิ่มได้)'); return; } const btn = document.getElementById('spinBtn'); btn.disabled = true; // 1. สุ่มเลือก Index ของช่องตรงๆ (0 ถึง 19) const targetIndex = Math.floor(Math.random() * wheelValues.length); const targetToken = wheelValues[targetIndex]; // 2. คำนวณองศาให้เข็มด้านบน (ชี้ที่ 0 องศา / 360 องศา) ไปหยุดกลางช่องที่สุ่มได้พอดี const sliceAngle = 360 / 20; // 18 องศาต่อช่อง // คำนวณกึ่งกลางของช่องเป้าหมาย const targetCenterAngle = (targetIndex * sliceAngle) + (sliceAngle / 2); // เนื่องจากวงล้อหมุนตามเข็ม การจะให้ช่องเป้าหมายมาอยู่ข้างบน (0 องศา) ต้องหมุนลบด้วยมุมของเป้าหมาย const degreesToRotate = 360 - targetCenterAngle; // บวกจำนวนรอบหมุนเพิ่ม (เช่น 6 รอบเต็ม) เพื่อความสมจริง const totalRotation = currentRotation - (currentRotation % 360) + (360 * 6) + degreesToRotate; currentRotation = totalRotation; const wheel = document.getElementById('wheel'); wheel.style.transform = `rotate(${currentRotation}deg)`; // 3. แสดงผลลัพธ์หลังจากวงล้อหยุดหมุน (4 วินาทีเท่ากับ transition ใน CSS) setTimeout(() => { userTokens += targetToken; localStorage.setItem('exotic_tokens', userTokens); localStorage.setItem('exotic_free_spin', 'used'); freeSpinsToday = false; document.getElementById('userTokenCount').innerText = userTokens; updateSpinButtonUI(); btn.disabled = false; alert(`🎉 ยินดีด้วยครับ! คุณสุ่มได้ ${targetToken} Token! บันทึกลงกระเป๋าเรียบร้อยแล้ว`); }, 4000); }
